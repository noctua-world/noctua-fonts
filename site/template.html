    type Err = ParseIntError;

    /// {{code.rs_c2}}
    fn from_str(s: &str) -> Result<Self, Self::Err> {
        let hex = s.trim_start_matches('#');
        let byte = |i: usize| u8::from_str_radix(&hex[i..i + 2], 16);
        Ok(Rgb { r: byte(0)?, g: byte(2)?, b: byte(4)? })
    }
}

    /** {{code.ts_c1}} */
export function debounce<T extends unknown[]>(fn: (...args: T) => void, ms = 120) {
  let timer: ReturnType<typeof setTimeout> | undefined;
  return (...args: T): void => {
    if (timer !== undefined) clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}

def audit(path: Path) -> dict[str, int]:
    text = path.read_text(encoding="utf-8")
    counts = Counter(ch for ch in text if ch in CONFUSABLE)
    return {ch: n for ch, n in counts.most_common() if n >= 1}
